# Apply the Banner to SSH Logins
sudo vi /etc/ssh/sshd_config  #Open the SSH configuration file
Banner /etc/issue.net         #Find the Banner directive or add it if it does not exist
sudo systemctl restart ssh    #Save the file and restart the SSH service

# Apply the Banner to Local Console Logins
sudo cp /etc/issue.net /etc/issue  #To display the same banner for local console logins
